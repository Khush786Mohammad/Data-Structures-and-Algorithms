class Solution {
    public int findMin(int[] nums) {
        int minimum = Integer.MAX_VALUE;
        int low = 0;
        int high = nums.length-1;

        while(low <= high){
            int mid = low + (high - low)/2;

            //means left half is sorted and the low will be the minimm of left half 
            if(nums[mid] >= nums[low]){
                minimum = Math.min(minimum, nums[low]);
                low = mid+1;
            } 
            else{
                // right half is sorted the minimum will be the mid
                minimum = Math.min(minimum, nums[mid]);
                high = mid-1;
            }
        }
        return minimum;
    }
}
