class Solution {
    public int findFirstOcc(int[] nums, int low, int high, int k){
        int idx = -1;
        while(low <= high){
            int mid = low + (high-low)/2;
            if(nums[mid] == k){
                idx = mid;
                high = mid-1;
            }
            else if(nums[mid] > k)
                high = mid-1;
            else 
                low = mid+1;
        }
        return idx;
    }
        public int findSecondOcc(int[] nums, int low, int high, int k){
        int idx = -1;
        while(low <= high){
            int mid = low + (high-low)/2;
            if(nums[mid] == k){
                idx = mid;
                low = mid+1;
            }
            else if(nums[mid] > k)
                high = mid-1;
            else 
                low = mid+1;
        }
        return idx;
    }
    public int[] searchRange(int[] nums, int target) {
        int[] ans = {-1,-1};
        int first = findFirstOcc(nums, 0 , nums.length-1, target);
        if(first == -1)
            return ans;
        int second = findSecondOcc(nums, 0 , nums.length-1, target);
        ans[0] = first;
        ans[1] = second;
        return ans;
    }
}
