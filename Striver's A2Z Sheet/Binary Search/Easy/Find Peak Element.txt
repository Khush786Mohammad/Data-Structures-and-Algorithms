class Solution {
    private int peakIdx(int[] nums, int low, int high){
        while(low <= high){
            int mid = low + (high - low)/2;
            if(nums[mid] > nums[mid-1] && nums[mid] > nums[mid+1])
                return mid;
            else if(nums[mid] > nums[mid-1])
                low = mid+1;
            else    
                high = mid-1;
        }
        return low;
    }
    public int findPeakElement(int[] nums) {
        int n = nums.length;
        if(n == 1 || nums[0] > nums[1])
            return 0;
        if(nums[n-1] > nums[n-2])
            return n-1;
        return peakIdx(nums, 1, n-2);
    }
}
