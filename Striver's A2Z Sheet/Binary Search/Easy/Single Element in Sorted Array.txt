class Solution {
    private int nonRepeating(int[] nums, int low, int high){
        while(low <= high){
            int mid = low + (high-low)/2;
            if(mid%2 == 0){
                if(mid+1 < nums.length && nums[mid] == nums[mid+1])
                    low = mid+1;
                else
                    high = mid-1;
            }
            else{
                if(nums[mid] == nums[mid-1])
                    low = mid+1;
                else
                    high = mid-1;
            }
        }
        return nums[high+1];
    }
    public int singleNonDuplicate(int[] nums) {
        int n = nums.length;
        if(n == 1)
            return nums[0];
        return nonRepeating(nums, 0, n-1);
    }
}
