class Solution {
    public int shipWithinDays(int[] weights, int days) {
        int n = weights.length;
        int minWght = Arrays.stream(weights).max().getAsInt();
        int maxWght = Arrays.stream(weights).sum();
        int ans = 0;

        while(minWght <= maxWght){
            int threshold = minWght + (maxWght-minWght)/2;
            int currWght = 0;
            int totalDays = 1;
            for(int i = 0 ; i<n ; i++){
                if(currWght + weights[i] <= threshold)
                    currWght += weights[i];
                else{
                    totalDays++;
                    currWght = weights[i];
                }
            }
            if(totalDays <= days){
                ans = threshold;
                maxWght = threshold-1;
            }
            else
                minWght = threshold+1;
        }

        return ans;
    }
}
