class Solution {
    public int smallestDivisor(int[] nums, int threshold) {
        int ans = 0;
        int n = nums.length;

        int minDivisor = 1;
        int maxDivisor = Arrays.stream(nums).max().getAsInt();
    
        while(minDivisor <= maxDivisor){
            int divisor = minDivisor + (maxDivisor-minDivisor)/2;
            int totalThreshold = 0;
            for(int i = 0 ; i<n ; i++){
                totalThreshold += nums[i]/divisor;
                if(nums[i]%divisor != 0)
                    totalThreshold += 1;
            }
            if(totalThreshold <= threshold){
                ans = divisor;
                maxDivisor = divisor-1;
            }
            else{
                minDivisor = divisor+1;
            }
        }
        return ans;
    }
}
