class Solution {
    public double calculatePower(double x, int n){
        if(n == 0)
            return 1.0;
        if(n == 1)
            return x;
        double result = calculatePower(x, n/2);
        if(n%2 == 0)
            return result * result;
        return x * result * result;
    }
    public double myPow(double x, int n) {
        double result = calculatePower(x, n);
        if(n >= 0)
            return result;
        return 1/result;
    }
}
