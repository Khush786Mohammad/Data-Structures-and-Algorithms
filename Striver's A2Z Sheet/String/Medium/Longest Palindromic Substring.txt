class Solution {
    public String longestPalindrome(String s) {
        int n = s.length();
        int start = 0;
        int end = 0;
        int maxLen = 0;

        for(int i = 0 ;i<n ;i++){
            // for even palindromic substring length
            int l = i;
            int r = i+1;
            while(l >= 0 && r<=n-1 && s.charAt(l) == s.charAt(r)){
                if(r-l+1 > maxLen){
                    maxLen = r-l+1;
                    start = l;
                    end = r;
                }
                l--;
                r++;
            }
            // for odd palindromic substring length
            l = i;
            r = i;
            while(l >= 0 && r<=n-1 && s.charAt(l) == s.charAt(r)){
                if(r-l+1 > maxLen){
                    maxLen = r-l+1;
                    start = l;
                    end = r;
                }
                l--;
                r++;
            }
        }
        return s.substring(start, end+1);
    }
}
