class Solution {
    private:
    int NcR(int n, int r){
        /*
            n->10, r->3
            nCr = 10!
                ------- since 7! of numerator will cancel the 7! of denominator so we need to
                3! * 7! find only 10*9*8

            10 * 9 * 8
           ------------   numerator will be n-i/i, since 10/3 will result the value in decimal
            3  * 2 * 1    so we will find 10/1 so that 10/1 * 9/2 * 8/3 will always result in
                          integer value

            10 * 9 * 8
           ------------
            1  * 2 * 3
        */

        int res = 1;
        for(int i = 0; i<r ; i++){
            res = res * (n-i);
            res = res/(i+1);
        }
        return res;
    }
public:
    vector<vector<int>> generate(int numRows) {
        vector<vector<int>> ans;
        for(int i = 0; i<numRows; i++){
            vector<int> result;
            for(int j = 0; j<=i ; j++){
                result.push_back(NcR(i,j));
            }
            ans.push_back(result);
        }
        return ans;
    }
};
