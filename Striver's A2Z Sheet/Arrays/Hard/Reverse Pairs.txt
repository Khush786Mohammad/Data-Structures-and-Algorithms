typedef long long ll;
class Solution {
    private:
    int divideNConquer(vector<int>& nums, int low, int high){
        int ans = 0;
        if(low < high){
            int mid = low + (high-low)/2;
            ans += divideNConquer(nums,low,mid);
            ans += divideNConquer(nums,mid+1,high);
            ans += merge(nums,low,mid,high);
        }
        return ans;
    }
    int merge(vector<int>& nums, int low, int mid, int high){
        vector<int> temp;
        int i = low;
        int j = mid+1;
        int ans = 0;

        while(i<=mid && j<=high){
            ll value = 2ll * nums[j];
            if(nums[i] > value){
                ans += mid-i+1;
                j++;
            }
            else
                i++;
        }
        i = low;
        j = mid+1;
        while(i<=mid && j<=high){
            if(nums[i] < nums[j]){
                temp.push_back(nums[i]);
                i++;
            }
            else{
                temp.push_back(nums[j]);
                j++;
            }
        }
        while(i<=mid){
            temp.push_back(nums[i]);
            i++;
        }
        while(j<=high){
            temp.push_back(nums[j]);
            j++;
        }
        for(int l = 0, q=low; l<high-low+1; l++,q++){
            nums[q] = temp[l];
        }
        return ans;
    }
public:
    int reversePairs(vector<int>& nums) {
        return divideNConquer(nums, 0, nums.size()-1);
    }
};
