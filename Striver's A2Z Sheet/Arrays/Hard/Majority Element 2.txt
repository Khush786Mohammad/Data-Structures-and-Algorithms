class Solution {
  public:
    vector<int> findMajority(vector<int>& nums) {
        int n = nums.size();
        
        int val1 = nums[0];
        int cnt1 = 1;
        int val2 = nums[0];
        int cnt2 = 1;
        
        for(int i = 1; i<n; i++){
            if(nums[i] == val1)
                cnt1++;
            else if(nums[i] == val2)
                cnt2++;
            else if(val1 == val2){
                val2 = nums[i];
                cnt2 = 1;
            }
            else{
                if(cnt1 == 0 && nums[i] != val1){
                    val1 = nums[i];
                    cnt1 = 1;
                }
                else if(cnt2 == 0 && nums[i] != val2){
                    val2 = nums[i];
                    cnt2 = 1;
                }
                else{
                    cnt1--;
                    cnt2--;
                }
            }
        }
        
        cnt1 = cnt2 = 0;
        for(int i = 0 ; i<n ; i++){
            if(nums[i] == val1)
                cnt1++;
            if(nums[i] == val2)
                cnt2++;
        }
        vector<int> ans;
        if(cnt1 > n/3)
            ans.push_back(val1);
        if(cnt2 > n/3 && val1 != val2)
            ans.push_back(val2);
            
        sort(ans.begin(), ans.end());
        return ans;
    }
};
