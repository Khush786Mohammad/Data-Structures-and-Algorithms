class Solution {
  public:
    int longestSubarray(vector<int>& arr, int k) {
        int n = arr.size();
        unordered_map<int,int> mp;
        int sum = 0;
        int ans = 0;
        
        for(int i = 0 ; i<n ; i++){
            sum += arr[i];
            if(sum == k)
                ans = max(ans,i+1);
            else{
                int req = sum - k;
                if(mp.find(req) != mp.end()){
                    ans = max(ans, i-mp[req]);
                }
            }
            if(mp.find(sum) == mp.end())
                mp[sum] = i;
        }
        return ans;
    }
};
