class Solution {
public:
    void sortColors(vector<int>& nums) {
        //Dutch National Flag Algorithm
        int n = nums.size();
        int low = 0;
        int mid = 0;
        int high = n-1;

        while(mid <= high){
            //swap nums[low] & nums[mid] increment both pointers
            if(nums[mid] == 0){
                swap(nums[low], nums[mid]);
                low++;
                mid++;
            }
            else if(nums[mid] == 1)
                mid++;
            else{
                swap(nums[mid], nums[high]);
                high--;
            }
        }
        return;
    }
};
