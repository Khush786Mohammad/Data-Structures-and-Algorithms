class Solution {
public:
    vector<int> rearrangeArray(vector<int>& nums) {
        int n = nums.size();
        vector<int> arr(n,0);
        int posIdx = 0;
        int negIdx = 1;
        int i = 0;
        int j = 0;
        
        while(i<n && j<n){
            while(i<n && nums[i] < 0)
                i++;
            while(j<n && nums[j] >= 0)
                j++;
            arr[posIdx] = nums[i++];
            arr[negIdx] = nums[j++];

            posIdx += 2;
            negIdx += 2;
        }
        return arr;
    }
};
