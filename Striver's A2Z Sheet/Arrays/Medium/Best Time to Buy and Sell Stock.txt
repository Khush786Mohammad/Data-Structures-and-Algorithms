class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int n = prices.size();
        int ans = 0;
        int buyDay = 0;

        for(int i = 1 ; i<n ; i++){
            if(prices[i] > prices[buyDay])
                ans = max(ans, prices[i] - prices[buyDay]);
            buyDay = prices[i] < prices[buyDay] ? i : buyDay;
        }
        return ans;
    }
};
