class Solution {
public:
    void setZeroes(vector<vector<int>>& matrix) {
        int row = matrix.size();
        int col = matrix[0].size();

        vector<int> horizontal(col, 1);
        vector<int> vertical(row, 1);

        for(int i = 0 ; i<row; i++){
            for(int j = 0 ; j<col ; j++){
                if(matrix[i][j] == 0){
                    vertical[i] = 0;
                    horizontal[j] = 0;
                }
            }
        }

        for(int i = 0 ; i<row ; i++){
            for(int j = 0; j<col ; j++){
                if(vertical[i] == 0 || horizontal[j] == 0){
                    matrix[i][j] = 0;
                }
            }
        }
    }
};
