class Solution {
public:
    vector<int> leaders(vector<int>& nums) {
      vector<int> ans;
      int n = nums.size();
      int max = INT_MIN;

      for(int i = n-1; i>=0 ; i--){
        if(nums[i] >= max){
            max = nums[i];
            ans.push_back(max);
        }
      }
      reverse(ans.begin(), ans.end());
      return ans;
    }
};
